{{- /* Image render hook. Path handling: content files store image paths RELATIVE to the markdown file (../static/images/foo.png) so editor previews resolve them; Hugo rewrites them here to the site-absolute /images/ that static/ serves. Site-absolute /images/... destinations pass through unchanged. Alignment: an optional URL fragment selects a class — #center -> img-center (block, horizontally centred) #left -> img-left (float left, text wraps) #right -> img-right (float right, text wraps) The fragment is consumed here, never emitted into src. Styles live in assets/css/extended/img-align.css. */ -}} {{- $dest := .Destination -}} {{- $align := "" -}} {{- if strings.Contains $dest "#" -}} {{- $parts := split $dest "#" -}} {{- $dest = index $parts 0 -}} {{- $frag := lower (index $parts 1) -}} {{- if in (slice "center" "centre" "left" "right") $frag -}} {{- $align = printf " class=\"img-%s\"" (cond (eq $frag "centre") "center" $frag) -}} {{- end -}} {{- end -}} {{- if strings.Contains $dest "static/images/" -}} {{- $dest = printf "/images/%s" (index (split $dest "static/images/") 1) -}} {{- else if strings.Contains $dest "static/pdfs/" -}} {{- $dest = printf "/pdfs/%s" (index (split $dest "static/pdfs/") 1) -}} {{- end -}}
